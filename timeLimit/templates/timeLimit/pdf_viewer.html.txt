    <!-- PDF.JS -->
    <script src="https://cdnjs.cloudflare.com/ajax/libs/pdf.js/4.10.38/pdf.min.mjs"
            type="module"></script>


    <script type="module">

        import * as pdfjsLib
            from "https://cdnjs.cloudflare.com/ajax/libs/pdf.js/4.10.38/pdf.min.mjs";


        /*
            PDF.js Worker
        */

        pdfjsLib.GlobalWorkerOptions.workerSrc =
            "https://cdnjs.cloudflare.com/ajax/libs/pdf.js/4.10.38/pdf.worker.min.mjs";


        /*
            PDF URL

            Later our Django detail page will pass
            the selected PDF URL here.
        */

        const params = new URLSearchParams(
            window.location.search
        );

        const pdfUrl =
            params.get("url");


        /*
            Elements
        */

        const canvas =
            document.getElementById("pdfCanvas");

        const ctx =
            canvas.getContext("2d");

        const loading =
            document.getElementById("loading");

        const errorMessage =
            document.getElementById("errorMessage");

        const pageNumberInput =
            document.getElementById("pageNumber");

        const pageCount =
            document.getElementById("pageCount");

        const zoomValue =
            document.getElementById("zoomValue");

        const prevPage =
            document.getElementById("prevPage");

        const nextPage =
            document.getElementById("nextPage");

        const zoomIn =
            document.getElementById("zoomIn");

        const zoomOut =
            document.getElementById("zoomOut");

        const fitPage =
            document.getElementById("fitPage");

        const downloadPdf =
            document.getElementById("downloadPdf");

        const printPdf =
            document.getElementById("printPdf");


        /*
            PDF variables
        */

        let pdfDoc = null;

        let currentPage = 1;

        let scale = 1.0;

        let rendering = false;

        let pendingPage = null;


        /*
            Render Page
        */

        async function renderPage(pageNum) {

            if (!pdfDoc) {
                return;
            }

            rendering = true;

            const page =
                await pdfDoc.getPage(pageNum);


            const viewport =
                page.getViewport({
                    scale: scale
                });


            canvas.width =
                viewport.width;

            canvas.height =
                viewport.height;


            const renderContext = {
                canvasContext: ctx,
                viewport: viewport
            };


            await page.render(
                renderContext
            ).promise;


            currentPage =
                pageNum;


            pageNumberInput.value =
                currentPage;


            pageNumberInput.max =
                pdfDoc.numPages;


            pageCount.textContent =
                "/ " + pdfDoc.numPages;


            zoomValue.textContent =
                Math.round(scale * 100) + "%";


            prevPage.disabled =
                currentPage <= 1;


            nextPage.disabled =
                currentPage >= pdfDoc.numPages;


            rendering = false;


            if (pendingPage !== null) {

                const next =
                    pendingPage;

                pendingPage = null;

                renderPage(next);
            }
        }


        /*
            Go to Page
        */

        function queueRenderPage(pageNum) {

            if (rendering) {

                pendingPage =
                    pageNum;

            } else {

                renderPage(pageNum);
            }
        }


        /*
            Previous
        */

        prevPage.addEventListener(
            "click",
            function() {

                if (
                    currentPage <= 1
                ) {
                    return;
                }

                queueRenderPage(
                    currentPage - 1
                );
            }
        );


        /*
            Next
        */

        nextPage.addEventListener(
            "click",
            function() {

                if (
                    !pdfDoc ||
                    currentPage >= pdfDoc.numPages
                ) {
                    return;
                }

                queueRenderPage(
                    currentPage + 1
                );
            }
        );


        /*
            Page Number
        */

        pageNumberInput.addEventListener(
            "change",
            function() {

                let page =
                    parseInt(
                        pageNumberInput.value,
                        10
                    );

                if (
                    isNaN(page)
                ) {
                    page = 1;
                }

                page =
                    Math.max(
                        1,
                        Math.min(
                            page,
                            pdfDoc.numPages
                        )
                    );

                queueRenderPage(page);
            }
        );


        /*
            Zoom In
        */

        zoomIn.addEventListener(
            "click",
            function() {

                scale += 0.2;

                queueRenderPage(
                    currentPage
                );
            }
        );


        /*
            Zoom Out
        */

        zoomOut.addEventListener(
            "click",
            function() {

                scale -= 0.2;

                if (scale < 0.4) {
                    scale = 0.4;
                }

                queueRenderPage(
                    currentPage
                );
            }
        );


        /*
            Fit Page
        */

        fitPage.addEventListener(
            "click",
            async function() {

                if (!pdfDoc) {
                    return;
                }

                const page =
                    await pdfDoc.getPage(
                        currentPage
                    );


                const viewport =
                    page.getViewport({
                        scale: 1
                    });


                const container =
                    document.getElementById(
                        "viewerContainer"
                    );


                const availableWidth =
                    container.clientWidth - 40;


                const availableHeight =
                    container.clientHeight - 40;


                const widthScale =
                    availableWidth /
                    viewport.width;


                const heightScale =
                    availableHeight /
                    viewport.height;


                scale =
                    Math.min(
                        widthScale,
                        heightScale
                    );


                queueRenderPage(
                    currentPage
                );
            }
        );


        /*
            Keyboard shortcuts
        */

        document.addEventListener(
            "keydown",
            function(event) {

                if (
                    event.key === "ArrowLeft"
                ) {
                    prevPage.click();
                }

                if (
                    event.key === "ArrowRight"
                ) {
                    nextPage.click();
                }

                if (
                    event.key === "+"
                ) {
                    zoomIn.click();
                }

                if (
                    event.key === "-"
                ) {
                    zoomOut.click();
                }
            }
        );


        /*
            Download
        */

        downloadPdf.addEventListener(
            "click",
            function() {

                if (!pdfUrl) {
                    return;
                }

                window.open(
                    pdfUrl,
                    "_blank"
                );
            }
        );


        /*
            Print

            Opens the original PDF in a new tab
            so browser print controls can be used.
        */

        printPdf.addEventListener(
            "click",
            function() {

                if (!pdfUrl) {
                    return;
                }

                const printWindow =
                    window.open(
                        pdfUrl,
                        "_blank"
                    );

                if (printWindow) {

                    printWindow.focus();

                    setTimeout(
                        function() {
                            printWindow.print();
                        },
                        1500
                    );
                }
            }
        );


        /*
            Load PDF
        */

        async function loadPDF() {

            if (!pdfUrl) {

                loading.style.display =
                    "none";

                errorMessage.style.display =
                    "flex";

                errorMessage.textContent =
                    "PDF URL नहीं मिला।";

                return;
            }


            try {

                const loadingTask =
                    pdfjsLib.getDocument({
                        url: pdfUrl
                    });


                pdfDoc =
                    await loadingTask.promise;


                pageCount.textContent =
                    "/ " + pdfDoc.numPages;


                pageNumberInput.max =
                    pdfDoc.numPages;


                loading.style.display =
                    "none";


                await renderPage(1);


            } catch (error) {

                console.error(
                    "PDF.js error:",
                    error
                );


                loading.style.display =
                    "none";


                errorMessage.style.display =
                    "flex";


                errorMessage.textContent =
                    "PDF preview नहीं खुल पाया। PDF को नए tab में खोलने के लिए left-side PDF link का उपयोग करें।";
            }
        }


        /*
            Start
        */

        loadPDF();

    </script>
